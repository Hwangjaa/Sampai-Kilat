### Reproduction commands

```bash
UA='Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Safari/537.36'
curl -s -A "$UA" 'https://fonts.googleapis.com/css2?family=Archivo:wght@400..800&display=swap'      # latin block -> archivo-latin.woff2
curl -s -A "$UA" 'https://fonts.googleapis.com/css2?family=IBM+Plex+Mono:wght@400;500&display=swap'  # 2x latin  -> ipm 400 / 500

# images (per file, same name/format, only if result < 85% of original)
ffmpeg -y -i IN.jpeg -vf "scale='min(1600,iw)':-2:flags=lanczos" -q:v 4 OUT.jpeg

# maps.png -> PNG-8 (dimensions preserved)
ffmpeg -y -i maps.png -vf "palettegen=max_colors=192:stats_mode=full" pal.png
ffmpeg -y -i maps.png -i pal.png -lavfi "paletteuse=dither=sierra2_4a" -pix_fmt pal8 maps8.png

# hero loop + poster
ffmpeg -y -i 'assets/homepage/video/FINAL CUT.mp4' -t 12 -vf "scale=1280:-2:flags=lanczos" \
  -c:v libx264 -crf 30 -preset slow -profile:v high -pix_fmt yuv420p -an \
  -movflags +faststart assets/video/hero-loop-720.mp4
ffmpeg -y -ss 1 -i assets/video/hero-loop-720.mp4 -frames:v 1 -vf "scale=1280:-2" -q:v 4 assets/video/hero-poster.jpg
```
